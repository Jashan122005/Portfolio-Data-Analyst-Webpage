-- which delivery lanes are slipping?
SELECT lane,
       COUNT(*) AS late_orders,
       AVG(delay_hrs) AS avg_delay
FROM deliveries
WHERE status = 'LATE'
GROUP BY lane
ORDER BY avg_delay DESC;
